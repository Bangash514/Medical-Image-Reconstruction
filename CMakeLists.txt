cmake_minimum_required(VERSION 3.10)
project(MedicalImageReconstruction CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenCV REQUIRED)
add_executable(reconstruct cpp/reconstruct.cpp)
target_include_directories(reconstruct PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(reconstruct ${OpenCV_LIBS})
target_compile_options(reconstruct PRIVATE -Wall -Wextra)
